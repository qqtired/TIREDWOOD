```js
import fs from 'node:fs';
import {beginRatFade,captureRatFadeFrame,endRatFade}
  from '/tmp/tiredwood-farm-pack1.NNujvo/./waterfront-rat-fade-callback.mjs';

// Для baseline null; для after читать baseline rat-fade-pose.json.
const poseBundle = stage === 'baseline' ? null :
  JSON.parse(fs.readFileSync(baselineDir + '/rat-fade-pose.json','utf8'));
const begun = await page.evaluate(beginRatFade,{poseBundle,climate:'clear',time:120});
const rows=[];
try {
  for(const distance of [45,60,70]) {
    const row=await page.evaluate(captureRatFadeFrame,{distance,settleMs:1800});
    rows.push(row);
    await page.screenshot({path:`${out}/rat-fade-${distance}m.png`});
  }
} finally {await page.evaluate(endRatFade);}
fs.writeFileSync(out+'/rat-fade-pose.json',JSON.stringify(begun.poseBundle));
fs.writeFileSync(out+'/rat-fade.json',JSON.stringify({begun,rows},null,2));
```
